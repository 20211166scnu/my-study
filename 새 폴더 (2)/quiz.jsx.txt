const CIRCLE = ["①","②","③","④","⑤","⑥"];

export default function App() {
  const total = QUESTIONS.length;
  const [idx, setIdx] = useState(0);
  const [selected, setSelected] = useState(null);
  const [revealed, setRevealed] = useState(false);
  const [hint, setHint] = useState(false);
  const [history, setHistory] = useState([]); // {q, topic, correct, skipped}
  const [finished, setFinished] = useState(false);

  const cur = QUESTIONS[idx];

  const answeredQs = history.map((h) => h.q);
  const correctCount = history.filter((h) => h.correct).length;

  function choose(i) {
    if (revealed) return;
    setSelected(i);
    setRevealed(true);
    setHistory((h) => [...h.filter((x) => x.q !== cur.q), { q: cur.q, topic: cur.topic, correct: i === cur.answer, skipped: false }]);
  }

  function skip() {
    if (!revealed) {
      setRevealed(true);
      setSelected(null);
      setHistory((h) => [...h.filter((x) => x.q !== cur.q), { q: cur.q, topic: cur.topic, correct: false, skipped: true }]);
    }
  }

  function next() {
    if (idx + 1 >= total) {
      setFinished(true);
      return;
    }
    setIdx(idx + 1);
    setSelected(null);
    setRevealed(false);
    setHint(false);
  }

  function restart() {
    setIdx(0); setSelected(null); setRevealed(false); setHint(false);
    setHistory([]); setFinished(false);
  }

  // ---------- Results ----------
  const topicStats = useMemo(() => {
    const m = {};
    history.forEach((h) => {
      if (!m[h.topic]) m[h.topic] = { total: 0, wrong: 0 };
      m[h.topic].total += 1;
      if (!h.correct) m[h.topic].wrong += 1;
    });
    return Object.entries(m).map(([topic, v]) => ({ topic, ...v }))
      .filter((t) => t.wrong > 0)
      .sort((a, b) => b.wrong - a.wrong);
  }, [history]);

  const pct = history.length ? Math.round((correctCount / history.length) * 100) : 0;

  return (
    <div style={S.page}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Gowun+Batang:wght@400;700&family=Gowun+Dodum&display=swap');
        * { box-sizing: border-box; -webkit-tap-highlight-color: transparent; }
        @keyframes fadeUp { from { opacity:0; transform:translateY(8px);} to {opacity:1;transform:none;} }
      `}</style>

      <div style={S.frame}>
        <header style={S.header}>
          <div style={S.kicker}>공중보건학 · INTERACTIVE QUIZ</div>
          <div style={S.brandRow}>
            <span style={S.brand}>문제 은행</span>
            <span style={S.count}>{finished ? `${total}/${total}` : `${idx + 1} / ${total}`}</span>
          </div>
          <div style={S.barTrack}>
            <div style={{ ...S.barFill, width: `${((finished ? total : idx) / total) * 100}%` }} />
          </div>
        </header>

        {!finished ? (
          <main key={cur.q} style={S.card}>
            <div style={S.qTopRow}>
              <span style={S.topicTag}>{cur.topic}</span>
              <span style={S.scoreMini}>맞힌 개수 {correctCount}</span>
            </div>
            <h1 style={S.qText}>{cur.q}. {cur.text}</h1>

            {hint && !revealed && (
              <div style={S.hintBox}>💡 힌트: 정답 보기는 “{cur.exp.split(" ").slice(0, 4).join(" ")}…”와 관련 있어요.</div>
            )}

            <div style={S.options}>
              {cur.options.map((opt, i) => {
                const isAnswer = i === cur.answer;
                const isPicked = i === selected;
                let st = { ...S.opt };
                if (revealed && isAnswer) st = { ...st, ...S.optCorrect };
                else if (revealed && isPicked && !isAnswer) st = { ...st, ...S.optWrong };
                else if (revealed) st = { ...st, ...S.optDim };
                return (
                  <button key={i} style={st} onClick={() => choose(i)} disabled={revealed}>
                    <span style={S.optCircle}>{CIRCLE[i]}</span>
                    <span style={S.optLabel}>{opt}</span>
                    {revealed && isAnswer && <span style={S.mark}>✓</span>}
                    {revealed && isPicked && !isAnswer && <span style={S.markX}>✕</span>}
                  </button>
                );
              })}
            </div>

            {revealed && (
              <div style={S.feedback}>
                <div style={S.verdict}>
                  {selected === null
                    ? <span style={{ color: "#8a6d3b" }}>건너뛰기 — 정답은 {CIRCLE[cur.answer]}</span>
                    : selected === cur.answer
                      ? <span style={{ color: "#3f6b4f" }}>정답입니다! ({CIRCLE[cur.answer]})</span>
                      : <span style={{ color: "#a8432f" }}>오답입니다. 정답은 {CIRCLE[cur.answer]}</span>}
                </div>
                <p style={S.exp}>{cur.exp}</p>
              </div>
            )}

            <div style={S.actions}>
              {!revealed && (
                <button style={S.ghostBtn} onClick={() => setHint(true)} disabled={hint}>힌트</button>
              )}
              {!revealed && (
                <button style={S.ghostBtn} onClick={skip}>건너뛰기</button>
              )}
              {revealed && (
                <button style={S.primaryBtn} onClick={next}>
                  {idx + 1 >= total ? "결과 보기" : "다음 문제 →"}
                </button>
              )}
            </div>
          </main>
        ) : (
          <main style={S.card}>
            <div style={S.resultHead}>QUIZ COMPLETE</div>
            <div style={S.bigScore}>{correctCount} <span style={S.bigScoreSub}>/ {total}</span></div>
            <div style={S.pct}>{pct}%</div>

            <div style={S.resultGrid}>
              <div style={S.statBox}><div style={S.statNum}>{correctCount}</div><div style={S.statLbl}>정답</div></div>
              <div style={S.statBox}><div style={S.statNum}>{history.filter((h)=>!h.correct && !h.skipped).length}</div><div style={S.statLbl}>오답</div></div>
              <div style={S.statBox}><div style={S.statNum}>{history.filter((h)=>h.skipped).length}</div><div style={S.statLbl}>건너뜀</div></div>
            </div>

            <h2 style={S.weakTitle}>자주 틀린 주제</h2>
            {topicStats.length === 0 ? (
              <p style={S.exp}>틀린 문제가 없어요. 완벽합니다! 🎉</p>
            ) : (
              <div style={S.weakList}>
                {topicStats.slice(0, 6).map((t) => (
                  <div key={t.topic} style={S.weakRow}>
                    <span style={S.weakName}>{t.topic}</span>
                    <span style={S.weakCount}>{t.wrong}문항 오답 / {t.total}문항</span>
                  </div>
                ))}
              </div>
            )}

            <button style={{ ...S.primaryBtn, marginTop: 24, width: "100%" }} onClick={restart}>다시 풀기</button>
          </main>
        )}

        <footer style={S.footer}>터치로 답을 선택하세요 · 힌트 / 건너뛰기 지원</footer>
      </div>
    </div>
  );
}
