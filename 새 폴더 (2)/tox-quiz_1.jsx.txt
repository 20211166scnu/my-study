function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

    setHint(false);

    <div style={S.page}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Black+Han+Sans&family=Gothic+A1:wght@400;500;700;800&family=JetBrains+Mono:wght@500;700&display=swap');
        *{box-sizing:border-box;-webkit-tap-highlight-color:transparent;}
        @keyframes pop{from{opacity:0;transform:translateY(10px);}to{opacity:1;transform:none;}}
        @keyframes glow{0%,100%{opacity:.5;}50%{opacity:1;}}
      `}</style>
      <div style={S.frame}>
        <header style={S.header}>
          <div style={S.topbar}>
            <div>
              <div style={S.kicker}>FOOD TOXICOLOGY</div>
              <div style={S.brand}>식품독성학 문제은행</div>
            </div>
            <button style={{ ...S.randomBtn, ...(random ? S.randomOn : {}) }} onClick={toggleRandom}>
              <span style={S.randomDot(random)} /> 랜덤 {random ? "ON" : "OFF"}
            </button>
          </div>
          <div style={S.metaRow}>
            <span style={S.counter}>{finished ? `${TOTAL}/${TOTAL}` : `${pos + 1} / ${TOTAL}`}</span>
            <span style={S.scoreChip}>● {correctCount} 정답 / {answeredN} 풀이</span>
          </div>
          <div style={S.barTrack}><div style={{ ...S.barFill, width: `${((finished ? TOTAL : pos) / TOTAL) * 100}%` }} /></div>
        </header>

        {!finished ? (
          <main key={qIdx} style={S.card}>
            <div style={S.qHead}>
              <span style={S.topicTag}>{cur.t}</span>
              <span style={S.typeTag}>{cur.type === "mc" ? "객관식" : "서술/복수"}</span>
            </div>
            <h1 style={S.qText}><span style={S.qNum}>Q{cur.n}.</span> {cur.text}</h1>

            {hint && !rec.revealed && (
              <div style={S.hintBox}>HINT · {cur.e.split(" ").slice(0, 5).join(" ")}…</div>
            )}

            {cur.type === "mc" ? (
              <div style={S.options}>
                {cur.o.map((opt, i) => {
                  const isAns = i === cur.a, isPick = i === rec.sel;
                  let st = { ...S.opt };
                  if (rec.revealed && isAns) st = { ...st, ...S.optCorrect };
                  else if (rec.revealed && isPick && !isAns) st = { ...st, ...S.optWrong };
                  else if (rec.revealed) st = { ...st, ...S.optDim };
                  return (
                    <button key={i} style={st} onClick={() => choose(i)} disabled={rec.revealed}>
                      <span style={S.optCircle}>{CIRCLE[i]}</span>
                      <span style={S.optLabel}>{opt}</span>
                      {rec.revealed && isAns && <span style={S.mark}>✓</span>}
                      {rec.revealed && isPick && !isAns && <span style={S.markX}>✕</span>}
                    </button>
                  );
                })}
              </div>
            ) : (
              <div style={S.saWrap}>
                {!rec.revealed ? (
                  <button style={S.revealBtn} onClick={revealSA}>정답 확인하기</button>
                ) : (
                  <div style={S.saAnsBox}>
                    <div style={S.saLabel}>정답</div>
                    <div style={S.saAns}>{cur.saAns}</div>
                  </div>
                )}
              </div>
            )}

            {rec.revealed && (
              <div style={S.feedback}>
                {cur.type === "mc" ? (
                  <div style={S.verdict}>
                    {rec.skipped ? <span style={{ color: "#d6a44a" }}>건너뜀 · 정답 {CIRCLE[cur.a]}</span>
                      : rec.sel === cur.a ? <span style={{ color: "#7ee0a1" }}>정답입니다 · {CIRCLE[cur.a]}</span>
                        : <span style={{ color: "#ff8a6b" }}>오답 · 정답은 {CIRCLE[cur.a]}</span>}
                  </div>
                ) : (
                  rec.self === undefined ? (
                    <div style={S.selfRow}>
                      <span style={S.selfPrompt}>스스로 채점하세요:</span>
                      <button style={{ ...S.selfBtn, ...S.selfOk }} onClick={() => selfGrade(true)}>맞음</button>
                      <button style={{ ...S.selfBtn, ...S.selfNo }} onClick={() => selfGrade(false)}>틀림</button>
                    </div>
                  ) : (
                    <div style={S.verdict}>{rec.self ? <span style={{ color: "#7ee0a1" }}>맞음으로 기록</span> : <span style={{ color: "#ff8a6b" }}>틀림으로 기록</span>}</div>
                  )
                )}
                <p style={S.exp}>{cur.e}</p>
              </div>
            )}

            <div style={S.actions}>
              <button style={{ ...S.navBtn, opacity: pos === 0 ? 0.35 : 1 }} onClick={prev} disabled={pos === 0}>← 이전</button>
              {!rec.revealed && cur.type === "mc" && <button style={S.ghostBtn} onClick={() => setHint(true)} disabled={hint}>힌트</button>}
              {!rec.revealed && cur.type === "mc" && <button style={S.ghostBtn} onClick={skip}>건너뛰기</button>}
              <button style={{ ...S.primaryBtn, opacity: canNext ? 1 : 0.4 }} onClick={next} disabled={!canNext}>
                {pos + 1 >= TOTAL ? "결과 보기" : "다음 →"}
              </button>
            </div>
          </main>
        ) : (
          <main style={S.card}>
            <div style={S.resultHead}>RESULT</div>
            <div style={S.bigScore}>{correctCount}<span style={S.bigSub}> / {TOTAL}</span></div>
            <div style={S.pct}>{pct}% {random && <span style={S.randTag}>(랜덤)</span>}</div>
            <div style={S.resultGrid}>
              <div style={S.statBox}><div style={S.statNum}>{correctCount}</div><div style={S.statLbl}>정답</div></div>
              <div style={S.statBox}><div style={S.statNum}>{answeredList.filter((r) => !r.correct && !r.skipped).length}</div><div style={S.statLbl}>오답</div></div>
              <div style={S.statBox}><div style={S.statNum}>{answeredList.filter((r) => r.skipped).length}</div><div style={S.statLbl}>건너뜀</div></div>
            </div>
            <h2 style={S.weakTitle}>자주 틀린 주제</h2>
            {topicStats.length === 0 ? (
              <p style={S.exp}>틀린 문제가 없어요. 완벽합니다.</p>
            ) : (
              <div style={S.weakList}>
                {topicStats.slice(0, 7).map((t) => (
                  <div key={t.t} style={S.weakRow}>
                    <span style={S.weakName}>{t.t}</span>
                    <span style={S.weakCount}>{t.wrong} / {t.total} 오답</span>
                  </div>
                ))}
              </div>
            )}
            <div style={{ display: "flex", gap: 10, marginTop: 22 }}>
              <button style={{ ...S.navBtn, flex: 1 }} onClick={prev}>← 마지막 문제</button>
              <button style={{ ...S.primaryBtn, flex: 2 }} onClick={restart}>다시 풀기</button>
            </div>
          </main>
        )}
        <footer style={S.footer}>터치 채점 · 이전 버튼으로 되돌아가기 · 상단에서 랜덤 출제 전환</footer>
      </div>
    </div>
  );
}
